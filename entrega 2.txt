-- 1. TABLA PARA TRIGGER DE SENTENCIA
-- ============================================================
CREATE TABLE auditoria_contrato (
    id_auditoria     NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    fecha_operacion  DATE DEFAULT SYSDATE,
    usuario          VARCHAR2(50) DEFAULT USER,
    tipo_operacion   VARCHAR2(10),
    filas_afectadas  NUMBER
);

-- ============================================================
-- 2. FUNCIONES
-- ============================================================
-- A. Con parametros: Deuda total arrendatario (RN6)
CREATE OR REPLACE FUNCTION fn_calcular_deuda_arrendatario(
    p_arrendatario_id NUMBER
) RETURN NUMBER IS
    v_deuda NUMBER := 0;
    v_existe NUMBER := 0;
BEGIN
    SELECT COUNT(*) INTO v_existe 
    FROM arrendatario 
    WHERE arrendatario_id = p_arrendatario_id;

    IF v_existe = 0 THEN
        RAISE_APPLICATION_ERROR(-20001, 'Arrendatario no existe');
    END IF;

    SELECT NVL(SUM(p.monto), 0)
    INTO v_deuda
    FROM pago_arriendo p
    JOIN contrato_arriendo c ON p.contrato_id = c.contrato_id
    WHERE c.arrendatario_id = p_arrendatario_id
      AND p.estado IN ('PENDIENTE', 'ATRASADO');

    RETURN v_deuda;
END;
/

-- B. Sin parametros: Total solicitudes tecnicas abiertas
CREATE OR REPLACE FUNCTION fn_total_solicitudes_abiertas
RETURN NUMBER IS
    v_total NUMBER := 0;
BEGIN
    SELECT COUNT(*)
    INTO v_total
    FROM solicitud_mantencion
    WHERE estado IN ('SOLICITADA', 'ASIGNADA', 'EN_PROCESO');

    RETURN v_total;
END;
/

-- ============================================================
-- 3. PROCEDIMIENTOS
-- ============================================================
-- A. Con parametros: Nuevo contrato atomico (RN1 con SAVEPOINT y ROLLBACK)
CREATE OR REPLACE PROCEDURE sp_registrar_contrato(
    p_unidad_id       NUMBER,
    p_arrendatario_id NUMBER,
    p_agente          VARCHAR2,
    p_inicio          DATE,
    p_termino         DATE,
    p_monto           NUMBER
) IS
    v_estado VARCHAR2(15);
BEGIN
    SAVEPOINT sp_inicio;

    IF p_termino <= p_inicio THEN
        RAISE_APPLICATION_ERROR(-20010, 'Fechas no validas');
    END IF;

    SELECT estado INTO v_estado
    FROM unidad
    WHERE unidad_id = p_unidad_id;

    IF v_estado <> 'DISPONIBLE' THEN
        RAISE_APPLICATION_ERROR(-20011, 'RN1: Unidad no esta disponible');
    END IF;

    INSERT INTO contrato_arriendo (
        unidad_id, arrendatario_id, agente_nombre, fecha_inicio, fecha_termino, monto_mensual, estado
    ) VALUES (
        p_unidad_id, p_arrendatario_id, p_agente, p_inicio, p_termino, p_monto, 'VIGENTE'
    );

    UPDATE unidad
    SET estado = 'ARRENDADA'
    WHERE unidad_id = p_unidad_id;

    COMMIT;
EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK TO sp_inicio;
        RAISE;
END;
/

-- B. Sin parametros: Reporte de contratos por estado
CREATE OR REPLACE PROCEDURE sp_reporte_contratos_estado IS
    CURSOR cur_contratos IS
        SELECT contrato_id, numero_unidad, nombre, estado, monto_mensual
        FROM contrato_arriendo c
        JOIN unidad u ON c.unidad_id = u.unidad_id
        JOIN arrendatario a ON c.arrendatario_id = a.arrendatario_id
        ORDER BY estado, contrato_id;
BEGIN
    FOR r IN cur_contratos LOOP
        DBMS_OUTPUT.PUT_LINE('[' || r.estado || '] Contrato: ' || r.contrato_id || 
                             ' | Unidad: ' || r.numero_unidad || 
                             ' | Arrendatario: ' || r.nombre || 
                             ' | Monto: $' || r.monto_mensual);
    END LOOP;
END;
/

-- ============================================================
-- 4. PACKAGE: GESTION DE AMENIDADES (RN4)
-- ============================================================
-- JUSTIFICACION MIEMBRO PRIVADO:
-- La funcion "fn_validar_solape" es privada para ocultar la logica de choque 
-- de horarios y evitar que se agenden reservas desde fuera sin pasar por las validaciones.

CREATE OR REPLACE PACKAGE pkg_gestion_amenidades AS
    PROCEDURE sp_agendar_reserva(
        p_amenidad NUMBER,
        p_arrendatario NUMBER,
        p_fecha DATE,
        p_inicio NUMBER,
        p_fin NUMBER
    );
    FUNCTION fn_total_reservas_dia(
        p_amenidad NUMBER,
        p_fecha DATE
    ) RETURN NUMBER;
END pkg_gestion_amenidades;
/

CREATE OR REPLACE PACKAGE BODY pkg_gestion_amenidades AS

    -- Miembro privado
    FUNCTION fn_validar_solape(
        p_amenidad NUMBER,
        p_fecha DATE,
        p_inicio NUMBER,
        p_fin NUMBER
    ) RETURN BOOLEAN IS
        v_cant NUMBER := 0;
    BEGIN
        SELECT COUNT(*) INTO v_cant
        FROM reserva_amenidad
        WHERE amenidad_id = p_amenidad
          AND TRUNC(fecha) = TRUNC(p_fecha)
          AND estado = 'CONFIRMADA'
          AND (hora_inicio < p_fin AND hora_fin > p_inicio);

        RETURN (v_cant > 0);
    END fn_validar_solape;

    -- Miembro publico 1
    PROCEDURE sp_agendar_reserva(
        p_amenidad NUMBER,
        p_arrendatario NUMBER,
        p_fecha DATE,
        p_inicio NUMBER,
        p_fin NUMBER
    ) IS
    BEGIN
        IF p_fin <= p_inicio THEN
            RAISE_APPLICATION_ERROR(-20020, 'Hora termino debe ser mayor');
        END IF;

        IF fn_validar_solape(p_amenidad, p_fecha, p_inicio, p_fin) THEN
            RAISE_APPLICATION_ERROR(-20021, 'RN4: Choque con otra reserva');
        END IF;

        INSERT INTO reserva_amenidad (
            amenidad_id, arrendatario_id, fecha, hora_inicio, hora_fin, estado
        ) VALUES (
            p_amenidad, p_arrendatario, TRUNC(p_fecha), p_inicio, p_fin, 'CONFIRMADA'
        );

        COMMIT;
    END sp_agendar_reserva;

    -- Miembro publico 2
    FUNCTION fn_total_reservas_dia(
        p_amenidad NUMBER,
        p_fecha DATE
    ) RETURN NUMBER IS
        v_total NUMBER := 0;
    BEGIN
        SELECT COUNT(*) INTO v_total
        FROM reserva_amenidad
        WHERE amenidad_id = p_amenidad
          AND TRUNC(fecha) = TRUNC(p_fecha)
          AND estado = 'CONFIRMADA';

        RETURN v_total;
    END fn_total_reservas_dia;

END pkg_gestion_amenidades;
/

-- 5. TRIGGERS
-- ============================================================
-- A. Trigger de sentencia: Cuenta filas afectadas
CREATE OR REPLACE TRIGGER trg_sentencia_auditoria_contrato
FOR INSERT OR UPDATE OR DELETE ON contrato_arriendo
COMPOUND TRIGGER
    v_tipo VARCHAR2(10);
    v_cont NUMBER := 0;

    AFTER EACH ROW IS
    BEGIN
        v_cont := v_cont + 1;
    END AFTER EACH ROW;

    AFTER STATEMENT IS
    BEGIN
        IF INSERTING THEN v_tipo := 'INSERT';
        ELSIF UPDATING THEN v_tipo := 'UPDATE';
        ELSIF DELETING THEN v_tipo := 'DELETE';
        END IF;

        IF v_cont > 0 THEN
            INSERT INTO auditoria_contrato (tipo_operacion, filas_afectadas)
            VALUES (v_tipo, v_cont);
        END IF;
    END AFTER STATEMENT;
END;
/

-- B. Trigger de fila 1: Libera unidad al finalizar contrato (RN2)
-- Reacciona tambien si se hace un UPDATE manual directo en la tabla
CREATE OR REPLACE TRIGGER trg_fila_liberar_unidad
AFTER UPDATE OF estado ON contrato_arriendo
FOR EACH ROW
WHEN (NEW.estado IN ('FINALIZADO', 'CANCELADO') AND OLD.estado = 'VIGENTE')
DECLARE
    v_activos NUMBER := 0;
BEGIN
    SELECT COUNT(*) INTO v_activos
    FROM contrato_arriendo
    WHERE unidad_id = :NEW.unidad_id
      AND estado = 'VIGENTE'
      AND contrato_id <> :NEW.contrato_id;

    IF v_activos = 0 THEN
        UPDATE unidad
        SET estado = 'DISPONIBLE'
        WHERE unidad_id = :NEW.unidad_id;
    END IF;
END;
/

-- C. Trigger de fila 2: Recalcula nivel de confianza (RN7)
CREATE OR REPLACE TRIGGER trg_fila_actualizar_confianza
AFTER INSERT ON movimiento_confianza
FOR EACH ROW
DECLARE
    v_puntos NUMBER := 0;
    v_nivel  NUMBER := 1;
BEGIN
    SELECT NVL(SUM(CASE WHEN tipo = 'POSITIVO' THEN puntos ELSE -puntos END), 0)
    INTO v_puntos
    FROM movimiento_confianza
    WHERE arrendatario_id = :NEW.arrendatario_id;

    SELECT nivel_id INTO v_nivel
    FROM (
        SELECT nivel_id
        FROM nivel_confianza
        WHERE puntos_minimos <= v_puntos
        ORDER BY puntos_minimos DESC
    )
    WHERE ROWNUM = 1;

    UPDATE arrendatario
    SET nivel_id = v_nivel
    WHERE arrendatario_id = :NEW.arrendatario_id;
END;
/
